function Save-TweakState {
    try {
        if (-not (Test-Path $Global:StateDir)) { New-Item -Path $Global:StateDir -ItemType Directory -Force | Out-Null }
        $json = @{ Orig = $Global:Orig; OrigServices = $Global:OrigServices; OrigMisc = $Global:OrigMisc; AppliedKeys = $Global:AppliedKeys } | ConvertTo-Json -Depth 10
        $json | Set-Content -Path $Global:StateFile -Encoding UTF8 -ErrorAction Stop
    } catch {}
}

function ConvertTo-HashtableDeep($obj) {
    if ($null -eq $obj) { return $null }
    if ($obj -is [System.Collections.IDictionary]) {
        $h = @{}; foreach ($k in $obj.Keys) { $h[$k] = ConvertTo-HashtableDeep $obj[$k] }; return $h
    }
    if ($obj -is [PSCustomObject]) {
        $h = @{}; foreach ($p in $obj.PSObject.Properties) { $h[$p.Name] = ConvertTo-HashtableDeep $p.Value }; return $h
    }
    if (($obj -is [System.Collections.IEnumerable]) -and ($obj -isnot [string])) {
        return @($obj | ForEach-Object { ConvertTo-HashtableDeep $_ })
    }
    return $obj
}

function Load-TweakState {
    try {
        if (Test-Path $Global:StateFile) {
            $rawText = Get-Content -Path $Global:StateFile -Raw -ErrorAction Stop
            $json = $null
            try { $json = $rawText | ConvertFrom-Json -ErrorAction Stop } catch {
                try {
                    $encoded = ""; foreach ($char in $rawText.ToCharArray()) { $encoded += [char]([int]$char -bxor 0x13) }
                    $bytes = [System.Convert]::FromBase64String($encoded)
                    $jsonStr = [System.Text.Encoding]::UTF8.GetString($bytes)
                    $json = $jsonStr | ConvertFrom-Json -ErrorAction Stop
                } catch {}
            }
            if ($json) {
                if ($json.Orig)          { $Global:Orig          = ConvertTo-HashtableDeep $json.Orig }
                if ($json.OrigServices)  { $Global:OrigServices  = ConvertTo-HashtableDeep $json.OrigServices }
                if ($json.OrigMisc)      { $Global:OrigMisc      = ConvertTo-HashtableDeep $json.OrigMisc }
                if ($json.AppliedKeys)   { $Global:AppliedKeys   = ConvertTo-HashtableDeep $json.AppliedKeys }
            }
        }
    } catch {}
}
Load-TweakState

# --- Helpers ---
function Get-RegValue($path, $name) {
    try {
        if (Test-Path $path) {
            $item = Get-ItemProperty -Path $path -Name $name -ErrorAction SilentlyContinue
            if ($null -ne $item) { return $item.$name }
        }
    } catch {}
    return $null
}

function Set-RegValue($path, $name, $value, $type = "DWord") {
    try {
        if (-not (Test-Path $path)) { New-Item -Path $path -Force | Out-Null }
        if ($type -eq "String") {
            Set-ItemProperty -Path $path -Name $name -Value $value -Type String -Force -ErrorAction Stop
        } else {
            Set-ItemProperty -Path $path -Name $name -Value $value -Type DWord -Force -ErrorAction Stop
        }
        return $true
    } catch { return $false }
}

function Save-Orig($key, $path, $name) {
    if (-not $Global:Orig.ContainsKey($key)) {
        $Global:Orig[$key] = @{ Path = $path; Name = $name; Value = (Get-RegValue $path $name) }
    }
}

function Save-OrigService($key, $svcName) {
    if (-not $Global:OrigServices.ContainsKey($key)) {
        try {
            $svc = Get-Service -Name $svcName -ErrorAction SilentlyContinue
            if ($svc) {
                $cim = Get-CimInstance Win32_Service -Filter "Name='$svcName'" -ErrorAction SilentlyContinue
                $Global:OrigServices[$key] = @{
                    ServiceName = $svcName
                    StartType   = if ($cim) { $cim.StartMode } else { $svc.StartType.ToString() }
                    Status      = $svc.Status.ToString()
                }
            }
        } catch {}
    }
}

function Restore-OrigService($info) {
    try {
        $name = $info.ServiceName
        if (-not $name) { return }
        $st = $info.StartType
        $wasRunning = ($info.Status -eq "Running")
        if ($st) {
            $mapped = switch ($st) { "Auto" { "Automatic" }; "Automatic" { "Automatic" }; "Manual" { "Manual" }; "Disabled" { "Disabled" }; default { "Manual" } }
            Set-Service -Name $name -StartupType $mapped -ErrorAction SilentlyContinue
        }
        if ($wasRunning) { Start-Service -Name $name -ErrorAction SilentlyContinue }
    } catch {}
}

function Add-Log([string]$msg, [string]$color = "#A3E635") {
    $ts = Get-Date -Format "HH:mm:ss"
    $line = "[$ts] $msg"
    $Global:LogLines.Add($line)
    if ($Global:LogBox) {
        $Global:LogBox.Dispatcher.Invoke([Action]{
            $Global:LogBox.Text += "$line`n"
            $Global:LogScroller.ScrollToEnd()
        })
    }
}

function Set-Status([string]$text) {
    if ($Global:StatusText) {
        $Global:StatusText.Dispatcher.Invoke([Action]{ $Global:StatusText.Text = $text })
    }
}

        $hasAmd    = [bool]($gpuNames -match "AMD|ATI|Radeon")

        if ($hasNvidia) {
            try {
                Stop-Service NvTelemetryContainer -Force -ErrorAction SilentlyContinue
                Set-Service NvTelemetryContainer -StartupType Disabled -ErrorAction SilentlyContinue
                Add-Log "NVIDIA telemetry disabled"
            } catch {}
            try {
                $path = "HKCU:\SOFTWARE\NVIDIA Corporation\Global\ShadowPlay\NVSPCAPS"
                Save-Orig "GPU_NvidiaOverlayOff" $path "value"
                Set-RegValue $path "value" 0 | Out-Null
                Add-Log "NVIDIA overlay disabled"
            } catch {}
        }
        if ($hasAmd) {
            try {
                Stop-Service "AMD External Events Utility Service" -Force -ErrorAction SilentlyContinue
                Set-Service "AMD External Events Utility Service" -StartupType Disabled -ErrorAction SilentlyContinue
                Add-Log "AMD External Events Utility disabled"
            } catch {}
            try {
                Stop-Service "AMD Crash Defender Service" -Force -ErrorAction SilentlyContinue
                Set-Service "AMD Crash Defender Service" -StartupType Disabled -ErrorAction SilentlyContinue
                Add-Log "AMD Crash Defender disabled"
            } catch {}
        }
    } catch {}

    $Global:AppliedKeys["ApplyAll"] = $true
    Save-TweakState
    Add-Log "=== APPLY COMPLETE ===" "#4ADE80"
    Add-Log "Some tweaks (BCD/HPET) need a reboot to fully take effect." "#FBBF24"
    Set-Status "READY"
}

# ============================================================
#  CLEAN
# ============================================================
function Invoke-Clean {
    Set-Status "CLEANING..."
    Add-Log "=== CLEAN ===" "#38BDF8"

    # Temp files
    try {
        $paths = @($env:TEMP, "$env:WINDIR\Temp")
        foreach ($p in $paths) {
            if (Test-Path $p) { Remove-Item -Path "$p\*" -Force -Recurse -ErrorAction SilentlyContinue }
        }
        Add-Log "Temp / junk files cleared"
    } catch { Add-Log "Temp clean partial" "#F87171" }

    # FiveM cache
    try {
        $p = "$env:LOCALAPPDATA\FiveM\FiveM.app\data\cache"
        if (Test-Path $p) {
            Remove-Item -Path "$p\*" -Force -Recurse -ErrorAction SilentlyContinue
            Add-Log "FiveM cache cleared"
        } else { Add-Log "FiveM cache not found - skipped" "#6B7280" }
    } catch {}

    # Shader cache
    try {
        $shaderPaths = @(
            "$env:LOCALAPPDATA\NVIDIA\DXCache", "$env:LOCALAPPDATA\NVIDIA\GLCache",
            "$env:LOCALAPPDATA\NVIDIA Corporation\NV_Cache",
            "$env:LOCALAPPDATA\AMD\DxCache", "$env:LOCALAPPDATA\AMD\DxcCache",
            "$env:LOCALAPPDATA\AMD\GLCache", "$env:LOCALAPPDATA\AMD\VkCache"
        )
        $found = 0
        foreach ($p in $shaderPaths) {
            if (Test-Path $p) {
                Remove-Item -Path "$p\*" -Force -Recurse -ErrorAction SilentlyContinue
                $found++
            }
        }
        if ($found -gt 0) { Add-Log "Shader cache cleared ($found folder(s))" }
        else { Add-Log "No shader cache found" "#6B7280" }
    } catch {}

    # Kill non-essential background processes
    $killList = @(
        "OneDrive","Spotify","SpotifyWebHelper","Discord","DiscordCanary","DiscordPTB","Skype","Teams",
        "SteamWebHelper","EpicGamesLauncher","EpicWebHelper","OriginWebHelperService","UbisoftConnectWebHelper",
        "GoogleCrashHandler","GoogleCrashHandler64","GoogleUpdate","AdobeUpdateService","AdobeIPCBroker",
        "OneDriveStandaloneUpdater","YourPhone","GameBar","GameBarFTServer","XboxAppServices",
        "NVIDIA GeForce Experience","AMDRSSrcExt","Dropbox","Cortana","SearchApp","WidgetService","Widgets"
    )
    $closed = 0
    foreach ($name in $killList) {
        try {
            $procs = Get-Process -Name $name -ErrorAction SilentlyContinue
            if ($procs) {
                $procs | Stop-Process -Force -ErrorAction SilentlyContinue
                $closed += @($procs).Count
            }
        } catch {}
    }
    Add-Log "Background processes closed: $closed"

    # Disable launcher startup
    $targets = @("Steam","Discord","Epic","Battle.net","Origin","Ubisoft","Uplay","Riot","GOG","Spotify")
    $runPaths = @(
        "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Run",
        "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Run"
    )
    $removed = 0
    $backup = @()
    foreach ($rp in $runPaths) {
        try {
            if (-not (Test-Path $rp)) { continue }
            $props = Get-ItemProperty -Path $rp -ErrorAction SilentlyContinue
            if (-not $props) { continue }
            foreach ($name in $props.PSObject.Properties.Name) {
                if ($name -in @("PSPath","PSParentPath","PSChildName","PSDrive","PSProvider")) { continue }
                foreach ($t in $targets) {
                    if ($name -match [regex]::Escape($t)) {
                        $val = $props.$name
                        $backup += @{ Path = $rp; Name = $name; Value = $val }
                        try {
                            Remove-ItemProperty -Path $rp -Name $name -ErrorAction Stop
                            $removed++
                            Add-Log "Startup disabled: $name"
                        } catch {}
                        break
                    }
                }
            }
        } catch {}
    }
    if ($backup.Count -gt 0) { $Global:OrigMisc["CLEAN_DisableLauncherStartup"] = $backup }
    if ($removed -gt 0) { Add-Log "Startup entries removed: $removed" }
    else { Add-Log "No matching startup entries found" "#6B7280" }

    # Prefetch / recent
    try {
        Remove-Item "$env:SystemRoot\Prefetch\*" -Force -ErrorAction SilentlyContinue
        Add-Log "Prefetch cleared"
    } catch {}

    Save-TweakState
    Add-Log "=== CLEAN COMPLETE ===" "#4ADE80"
    Set-Status "READY"
}

# ============================================================
#  RESTORE
# ============================================================
function Invoke-Restore {
    Set-Status "RESTORING..."
    Add-Log "=== RESTORE DEFAULTS ===" "#38BDF8"
    Load-TweakState

    $restored = 0

    # Restore registry values
    if ($Global:Orig -and $Global:Orig.Count -gt 0) {
        foreach ($key in @($Global:Orig.Keys)) {
            try {
                $info = $Global:Orig[$key]
                if ($null -eq $info) { continue }
                $path = $info.Path
                $name = $info.Name
                $val  = $info.Value
                if (-not $path -or -not $name) { continue }
                if ($null -eq $val) {
                    # Original was missing - remove the value we set
                    if (Test-Path $path) {
                        Remove-ItemProperty -Path $path -Name $name -ErrorAction SilentlyContinue
                    }
                } else {
                    $type = if ($val -is [string]) { "String" } else { "DWord" }
                    Set-RegValue $path $name $val $type | Out-Null
                }
                $restored++
            } catch {}
        }
        Add-Log "Registry values restored: $restored"
    } else {
        Add-Log "No saved registry originals found" "#6B7280"
    }

    # Restore services
    $svcCount = 0
    if ($Global:OrigServices -and $Global:OrigServices.Count -gt 0) {
        foreach ($key in @($Global:OrigServices.Keys)) {
            try {
                Restore-OrigService $Global:OrigServices[$key]
                $svcCount++
            } catch {}
        }
        Add-Log "Services restored: $svcCount"
    }

    # Restore startup entries
    if ($Global:OrigMisc.ContainsKey("CLEAN_DisableLauncherStartup")) {
        $backup = $Global:OrigMisc["CLEAN_DisableLauncherStartup"]
        $r = 0
        foreach ($entry in $backup) {
            try {
                if (-not (Test-Path $entry.Path)) { New-Item -Path $entry.Path -Force | Out-Null }
                Set-ItemProperty -Path $entry.Path -Name $entry.Name -Value $entry.Value -Force -ErrorAction Stop
                $r++
            } catch {}
        }
        Add-Log "Startup entries restored: $r"
    }

    # Remove FiveM QoS policies
    try {
        Get-NetQosPolicy -ErrorAction SilentlyContinue | Where-Object { $_.Name -like "SyncProject FiveM*" } |
            Remove-NetQosPolicy -Confirm:$false -ErrorAction SilentlyContinue
        Add-Log "FiveM QoS policies removed"
    } catch {}

    # Remove FiveM firewall rules
    try {
        Get-NetFirewallRule -ErrorAction SilentlyContinue | Where-Object { $_.DisplayName -like "SyncProject - FiveM*" } |
            Remove-NetFirewallRule -ErrorAction SilentlyContinue
        Add-Log "FiveM firewall rules removed"
    } catch {}

    # Remove IFEO for FiveM
    try {
        $ifeo = "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options\FiveM_GTAProcess.exe"
        if (Test-Path $ifeo) { Remove-Item -Path $ifeo -Recurse -Force -ErrorAction SilentlyContinue }
        Add-Log "FiveM IFEO priority entry removed"
    } catch {}

    # Reset common power / network defaults that are safe
    try {
        powercfg /setactive SCHEME_BALANCED 2>&1 | Out-Null
        Add-Log "Power plan set back to Balanced"
    } catch {}

    try {
        netsh int tcp set global autotuninglevel=normal 2>&1 | Out-Null
        Add-Log "TCP auto-tuning reset to normal"
    } catch {}

    # Clear state so next Apply starts fresh
    $Global:Orig = @{}
    $Global:OrigServices = @{}
    $Global:OrigMisc = @{}
    $Global:AppliedKeys = @{}
    Save-TweakState

    Add-Log "=== RESTORE COMPLETE ===" "#4ADE80"
    Add-Log "A reboot is recommended to fully clear BCD/timer changes." "#FBBF24"
    Set-Status "READY"
}

# Progress bar helper (animated fill during long ops - visual only)
function Set-ProgressBar([double]$pct) {
    if (-not $Global:ProgressFill) { return }
    $parent = $Global:ProgressFill.Parent
    $maxW = if ($parent -and $parent.ActualWidth -gt 0) { $parent.ActualWidth } else { 840 }
    $target = [Math]::Max(0, [Math]::Min(1, $pct)) * $maxW
    $anim = New-Object System.Windows.Media.Animation.DoubleAnimation($Global:ProgressFill.Width, $target, [TimeSpan]::FromMilliseconds(400))
    $anim.EasingFunction = New-Object System.Windows.Media.Animation.QuadraticEase
    $Global:ProgressFill.BeginAnimation([System.Windows.FrameworkElement]::WidthProperty, $anim)
}

# Hover scale
function Add-HoverScale($btn) {
    if (-not $btn) { return }
    $btn.Add_MouseEnter({
        $st = $this.RenderTransform
        if ($st -is [System.Windows.Media.ScaleTransform]) {
            $ax = New-Object System.Windows.Media.Animation.DoubleAnimation(1.0, 1.03, [TimeSpan]::FromMilliseconds(140))
            $ay = New-Object System.Windows.Media.Animation.DoubleAnimation(1.0, 1.03, [TimeSpan]::FromMilliseconds(140))
            $st.BeginAnimation([System.Windows.Media.ScaleTransform]::ScaleXProperty, $ax)
            $st.BeginAnimation([System.Windows.Media.ScaleTransform]::ScaleYProperty, $ay)
        }
    }.GetNewClosure())
    $btn.Add_MouseLeave({
        $st = $this.RenderTransform
        if ($st -is [System.Windows.Media.ScaleTransform]) {
            $ax = New-Object System.Windows.Media.Animation.DoubleAnimation($st.ScaleX, 1.0, [TimeSpan]::FromMilliseconds(160))
            $ay = New-Object System.Windows.Media.Animation.DoubleAnimation($st.ScaleY, 1.0, [TimeSpan]::FromMilliseconds(160))
            $st.BeginAnimation([System.Windows.Media.ScaleTransform]::ScaleXProperty, $ax)
            $st.BeginAnimation([System.Windows.Media.ScaleTransform]::ScaleYProperty, $ay)
        }
    }.GetNewClosure())
}

function Run-Safe([scriptblock]$action, [string]$label) {
    $Btn1.IsEnabled = $false; $Btn2.IsEnabled = $false; $Btn3.IsEnabled = $false
    Set-ProgressBar 0.08
    try {
        & $action
        Set-ProgressBar 1.0
        Start-Sleep -Milliseconds 350
        Set-ProgressBar 0
    } catch {
        Add-Log "ERROR during $label : $_" "#F87171"
        Set-Status "ERROR"
        Set-ProgressBar 0
    } finally {
        $Btn1.IsEnabled = $true; $Btn2.IsEnabled = $true; $Btn3.IsEnabled = $true
    }
}
